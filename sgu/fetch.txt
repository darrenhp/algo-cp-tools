async function batchTranslateSGU() {
    const start = 100;
    const end = 200;
    const delayMs = 5000; // 请求间隔，可自行调大

    for (let num = start; num <= end; num++) {
        const problemId = `SGU-${num}`;
        console.log(`===== 正在请求翻译: ${problemId} =====`);
        try {
            const res = await fetch("https://vjudge.net/problem/gptTranslate/request", {
                headers: {
                    "accept": "application/json, text/javascript, */*; q=0.01",
                    "accept-language": "zh,zh-CN;q=0.9",
                    "content-type": "application/x-www-form-urlencoded; charset=UTF-8",
                    "priority": "u=1, i",
                    "sec-ch-ua": "\"Chromium\";v=\"148\", \"Google Chrome\";v=\"148\", \"Not/A)Brand\";v=\"99\"",
                    "sec-ch-ua-mobile": "?0",
                    "sec-ch-ua-platform": "\"macOS\"",
                    "sec-fetch-dest": "empty",
                    "sec-fetch-mode": "cors",
                    "sec-fetch-site": "same-origin",
                    "x-requested-with": "XMLHttpRequest"
                },
                referrer: `https://vjudge.net/problem/${problemId}`,
                body: new URLSearchParams({
                    problem: problemId,
                    lang: "zh",
                    enforce: "true"
                }),
                method: "POST",
                mode: "cors",
                credentials: "include"
            });

            const data = await res.json();
            console.log(`${problemId} 翻译结果:`, data);
        } catch (err) {
            console.error(`${problemId} 请求失败:`, err);
        }
        // 延时
        await new Promise(r => setTimeout(r, delayMs));
    }
    console.log("全部任务遍历完成！");
}

// 启动批量翻译
batchTranslateSGU();
